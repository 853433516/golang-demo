package main

import (
	"bytes"
	"encoding/hex"
	"flag"
	"fmt"
	"math/big"
	"os"
	"os/signal"
	"strings"
	"syscall"
	"time"

	"github.com/fbsobreira/gotron-sdk/pkg/address"
	"github.com/fbsobreira/gotron-sdk/pkg/client"
	"github.com/fbsobreira/gotron-sdk/pkg/proto/api"
	"github.com/fbsobreira/gotron-sdk/pkg/proto/core"
	"google.golang.org/protobuf/proto"
)

// 简单示例：监听指定地址的 TRX/USDT 收款（基于区块轮询，解析 TransferContract 和 TRC20 transfer）
func main() {
	node := flag.String("node", "grpc.trongrid.io:50051", "gRPC node address")
	nodesFlag := flag.String("nodes", "", "备用节点列表，逗号分隔，优先级从左到右，例如 grpc.trongrid.io:50051,grpc.shasta.trongrid.io:50051")
	addrStr := flag.String("address", "", "目标接收地址（Base58）")
	poll := flag.Duration("poll", 3*time.Second, "轮询间隔")
	conf := flag.Int64("confirmations", 1, "需要多少个确认才认为支付成功")
	flag.Parse()

	if *addrStr == "" {
		_, _ = fmt.Fprintln(os.Stderr, "必须指定 --address 参数（Base58）")
		os.Exit(2)
	}

	targetAddr, err := address.Base58ToAddress(*addrStr)
	if err != nil {
		_, _ = fmt.Fprintln(os.Stderr, "address decode error:", err)
		os.Exit(1)
	}

	// 解析多节点配置（优先使用 --nodes 指定的列表）
	var nodes []string
	if *nodesFlag != "" {
		for _, n := range strings.Split(*nodesFlag, ",") {
			nn := strings.TrimSpace(n)
			if nn != "" {
				nodes = append(nodes, nn)
			}
		}
	}
	if len(nodes) == 0 {
		nodes = append(nodes, *node)
	}

	// 连接到首个可用节点（有简单重试）
	cur := 0
	var c *client.GrpcClient
	connect := func(idx int) error {
		if c != nil {
			c.Stop()
		}
		addr := nodes[idx]
		_, _ = fmt.Fprintf(os.Stderr, "connecting to node %s\n", addr)
		cc := client.NewGrpcClient(addr)
		cc.SetTimeout(30 * time.Second)
		if err := cc.Start(client.GRPCInsecure()); err != nil {
			return err
		}
		c = cc
		cur = idx
		return nil
	}

	// try connect first available
	var connErr error
	for i := 0; i < len(nodes); i++ {
		if err := connect(i); err == nil {
			connErr = nil
			break
		} else {
			connErr = err
			_, _ = fmt.Fprintf(os.Stderr, "connect attempt to %s failed: %v\n", nodes[i], err)
		}
	}
	if connErr != nil {
		_, _ = fmt.Fprintln(os.Stderr, "grpc start error (all nodes failed):", connErr)
		os.Exit(1)
	}
	defer func() {
		if c != nil {
			c.Stop()
		}
	}()

	// 获取当前区块高度，作为起点；采用重试以避免短暂的超时失败
	var now *api.BlockExtention
	var lastProcessed int64 = 0
	for i := 0; i < 5; i++ {
		n, err := c.GetNowBlock()
		if err == nil && n != nil && n.BlockHeader != nil {
			now = n
			lastProcessed = now.BlockHeader.RawData.Number - 1
			break
		}
		// 若遇到 RPC 超时或网络错误，尝试切换到下一个节点
		_, _ = fmt.Fprintf(os.Stderr, "get now block attempt %d failed: %v, retry in %s\n", i+1, err, time.Second)
		// 切换节点
		next := (cur + 1) % len(nodes)
		if next != cur {
			if err := connect(next); err == nil {
				_, _ = fmt.Fprintf(os.Stderr, "switched to node %s\n", nodes[next])
			} else {
				_, _ = fmt.Fprintf(os.Stderr, "switch to node %s failed: %v\n", nodes[next], err)
			}
		}
		time.Sleep(time.Second)
	}
	if now == nil {
		_, _ = fmt.Fprintln(os.Stderr, "get now block error: giving up after retries")
		os.Exit(1)
	}

	// 信号处理
	sig := make(chan os.Signal, 1)
	signal.Notify(sig, syscall.SIGINT, syscall.SIGTERM)

	ticker := time.NewTicker(*poll)
	defer ticker.Stop()

	fmt.Printf(" 开始监听 %s 的支付，起始区块 %d\n", *addrStr, lastProcessed+1)

	// 常见 TRC20 主网 USDT 合约（20 字节 hex），用于示例识别
	usdtHex := "a614f803b6fd780986a42c78ec9c7f77e6ded13c"
	usdtDecimals := 6

	for {
		select {
		case <-sig:
			fmt.Println("退出")
			return
		case <-ticker.C:
			tip, err := c.GetNowBlock()
			if err != nil {
				fmt.Println("get now block err:", err)
				continue
			}
			tipNum := tip.BlockHeader.RawData.Number
			// iterate new blocks
			for b := lastProcessed + 1; b <= tipNum-*conf; b++ {
				bext, err := c.GetBlockByNum(b)
				if err != nil {
					fmt.Println("get block error:", err)
					// 跳过当前区块，继续处理后续区块
					continue
				}
				// BlockExtention 结构包含 Transactions 和 BlockHeader
				if bext.GetBlockHeader() == nil {
					// older proto variants may differ; fallback
				}
				// api.BlockExtention has Transactions []*api.TransactionExtention
				for _, txext := range bext.GetTransactions() {
					if txext == nil || txext.Transaction == nil || txext.Transaction.RawData == nil {
						continue
					}
					// 检查交易返回（Ret）以获得交易执行结果，默认 UNKNOWN
					status := "UNKNOWN" // proto type in this environment may not expose Ret/ContractRet; keep UNKNOWN
					// 遍历合约
					for _, ctr := range txext.Transaction.RawData.Contract {
						if ctr == nil {
							continue
						}
						// native transfer (TRX)
						if ctr.Type == core.Transaction_Contract_TransferContract {
							var t core.TransferContract
							if ctr.Parameter != nil {
								if err := proto.Unmarshal(ctr.Parameter.Value, &t); err != nil {
									continue
								}
								if bytes.Equal(t.GetToAddress(), targetAddr.Bytes()) {
									// 格式化 TRX 数量（1 TRX = 1e6 sun）
									amt := uint64(t.GetAmount())
									human := formatBigUintAmount(amt, 6)
									fmt.Printf("检测到 TRX 收款 到 %s，数量: %s TRX，txid: %s，区块: %d，状态: %s\n",
										*addrStr, human, hex.EncodeToString(txext.Txid), b, status)
								}
							}
							continue
						}
						// TRC20: TriggerSmartContract -> 解析 data，看是否为 transfer(address,uint256)
						if ctr.Type == core.Transaction_Contract_TriggerSmartContract {
							var t core.TriggerSmartContract
							if ctr.Parameter != nil {
								if err := proto.Unmarshal(ctr.Parameter.Value, &t); err != nil {
									continue
								}
								data := t.GetData()
								// transfer selector: a9059cbb
								if len(data) >= 4 && bytes.Equal(data[0:4], []byte{0xa9, 0x05, 0x9c, 0xbb}) {
									// 32 bytes for to-address (data[4:36]), 接收方在最后 20 字节
									if len(data) >= 68 { // 4 + 32 + 32
										toWord := data[4:36]
										amountWord := data[36:68]
										// 提取 20 字节地址
										to20 := abiAddressTo20(toWord)
										// 规范化目标地址为 20 字节
										target20 := tronAddressTo20(targetAddr.Bytes())
										if bytes.Equal(to20, target20) {
											// 解析金额为 big.Int (uint256)
											amount := new(big.Int).SetBytes(amountWord)
											// 合约地址（token 合约）
											contractAddr := t.GetContractAddress()
											contract20 := tronAddressTo20(contractAddr)
											contractHex := hex.EncodeToString(contract20)
											isUSDT := strings.EqualFold(contractHex, usdtHex)
											var human string
											var tokenName string
											if isUSDT {
												human = formatBigAmount(amount, usdtDecimals)
												tokenName = "USDT"
											} else {
												// 对于未知代币，我们暂时直接显示原始整数，并把合约地址作为名字的一部分
												human = amount.String()
												tokenName = "TOKEN"
											}
											coin := "TRC20"
											if isUSDT {
												coin = "USDT(TRC20)"
											}
											fmt.Printf("检测到 %s 收款 到 %s，数量: %s %s，合约: 0x%s，txid: %s，区块: %d，状态: %s\n",
												coin, *addrStr, human, tokenName, contractHex, hex.EncodeToString(txext.Txid), b, status)
										}
									}
								}
							}
						}
					}
				}
				lastProcessed = b
			}
		}
	}
}

// abiAddressTo20 extracts the 20-byte address from a 32-byte ABI word (right-aligned)
func abiAddressTo20(word []byte) []byte {
	if len(word) == 32 {
		return word[12:32]
	}
	// fallback: if 20 bytes already
	if len(word) == 20 {
		return word
	}
	return nil
}

// tronAddressTo20 converts Tron address bytes (may be 21 bytes with 0x41 prefix) to 20-byte raw address
func tronAddressTo20(b []byte) []byte {
	if len(b) == 21 && b[0] == 0x41 {
		return b[1:21]
	}
	if len(b) == 20 {
		return b
	}
	// if longer/shorter, try to strip leading zeros to get last 20 bytes
	if len(b) > 20 {
		return b[len(b)-20:]
	}
	// unknown
	return b
}

// formatBigAmount formats a big.Int token amount into a decimal string using given decimals
func formatBigAmount(amount *big.Int, decimals int) string {
	if amount == nil {
		return "0"
	}
	if decimals <= 0 {
		return amount.String()
	}
	// bf = amount / 10^decimals
	bf := new(big.Float).SetInt(amount)
	denom := new(big.Int).Exp(big.NewInt(10), big.NewInt(int64(decimals)), nil)
	df := new(big.Float).SetInt(denom)
	bf.Quo(bf, df)
	// 保留 decimals 位小数
	return bf.Text('f', decimals)
}

// formatBigUintAmount formats a uint64 amount with decimals (e.g., TRX sun -> 6 decimals)
func formatBigUintAmount(v uint64, decimals int) string {
	bi := new(big.Int).SetUint64(v)
	return formatBigAmount(bi, decimals)
}
